cmake_minimum_required(VERSION 3.19)
# Version: MAJOR.MINOR.PATCH from VERSION (PATCH counts new features and bug
# fixes), with a fourth BUILD part when a build number is enabled through
# MLACKER_BUILD_NUMBER (cache variable or environment of the first configure).
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" MLACKER_BASE_VERSION LIMIT_COUNT 1)
if(NOT MLACKER_BASE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "VERSION must hold MAJOR.MINOR.PATCH, not '${MLACKER_BASE_VERSION}'")
endif()
set(MLACKER_BUILD_NUMBER "$ENV{MLACKER_BUILD_NUMBER}" CACHE STRING
    "Build number appended as a fourth version part (empty: none)")
set(MLACKER_VERSION "${MLACKER_BASE_VERSION}")
if(NOT MLACKER_BUILD_NUMBER STREQUAL "")
    if(NOT MLACKER_BUILD_NUMBER MATCHES "^[0-9]+$")
        message(FATAL_ERROR "MLACKER_BUILD_NUMBER must be a number, not '${MLACKER_BUILD_NUMBER}'")
    endif()
    string(APPEND MLACKER_VERSION ".${MLACKER_BUILD_NUMBER}")
endif()
project(mlacker VERSION ${MLACKER_VERSION} LANGUAGES C CXX OBJCXX)
message(STATUS "mlacker ${MLACKER_VERSION}")
if(NOT APPLE)
    message(FATAL_ERROR "mlacker currently requires macOS AUHAL")
endif()
if(NOT VST3_SDK_ROOT OR NOT MLACKER_MAIN_OBJECT)
    message(FATAL_ERROR "Use mlang pkg build, or specify VST3_SDK_ROOT and MLACKER_MAIN_OBJECT")
endif()
get_filename_component(VST3_SDK_ROOT "${VST3_SDK_ROOT}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(MLACKER_MAIN_OBJECT "${MLACKER_MAIN_OBJECT}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
# MLANG_ROOT: MLang checkout providing build/libmlang_std.a and stdlib/include.
if(NOT MLANG_ROOT)
    set(MLANG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/subprojects/mlang")
endif()
get_filename_component(MLANG_ROOT "${MLANG_ROOT}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_ADD_VST3_UTILITIES OFF CACHE BOOL "" FORCE)

set(SMTG_DISABLE_CODE_SIGNING ON CACHE BOOL "" FORCE)
set(SMTG_BUILD_UNIVERSAL_BINARY OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/sdk")
smtg_enable_vst3_sdk()
# Release packages link OpenSSL statically, so mlacker runs without Homebrew's
# OpenSSL (MLACKER_STATIC_OPENSSL=1 in the environment of the first configure).
set(MLACKER_STATIC_OPENSSL_DEFAULT OFF)
if("$ENV{MLACKER_STATIC_OPENSSL}")
    set(MLACKER_STATIC_OPENSSL_DEFAULT ON)
endif()
option(MLACKER_STATIC_OPENSSL "Link OpenSSL statically" ${MLACKER_STATIC_OPENSSL_DEFAULT})
if(MLACKER_STATIC_OPENSSL)
    set(OPENSSL_USE_STATIC_LIBS TRUE)
endif()
find_package(OpenSSL REQUIRED)
add_library(mlacker_host STATIC src/vst3_host.cpp
    "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp"
    "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/plugprovider.cpp"
    "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_mac.mm")
target_include_directories(mlacker_host PUBLIC "${MLANG_ROOT}/stdlib/include")
target_compile_options(mlacker_host PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_compile_definitions(mlacker_host PRIVATE MLACKER_VERSION="${MLACKER_VERSION}")
target_link_libraries(mlacker_host PUBLIC sdk_hosting "${MLANG_ROOT}/build/libmlang_std.a" OpenSSL::SSL OpenSSL::Crypto
    "-framework Cocoa" "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox"
    "-framework AudioUnit" "-framework CoreMIDI" "-framework AVFoundation" "-framework CoreGraphics" "-framework ImageIO")
set_source_files_properties("${MLACKER_MAIN_OBJECT}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(mlacker "${MLACKER_MAIN_OBJECT}")
set_target_properties(mlacker PROPERTIES LINKER_LANGUAGE CXX RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(mlacker PRIVATE mlacker_host)
if(EXISTS "${MLACKER_MAIN_OBJECT}.mlang-info.plist")
    target_link_options(mlacker PRIVATE "LINKER:-sectcreate,__TEXT,__info_plist,${MLACKER_MAIN_OBJECT}.mlang-info.plist")
endif()
